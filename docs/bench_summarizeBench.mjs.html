/**

/**
 * 材料之檢核(忠實性、修補):原文標題為首行(正式材料＝原文標題＋送出之內文)
 *
 * @param {Object} inp 輸入案之 input
 * @returns {Array} 回傳問題說明陣列
 */
function materialProblems(inp) {
    const p = []
    if (!isestr(inp.title)) p.push('缺 input.title(原文標題)')
    if (!isestr(inp.material)) p.push('缺 input.material(原文標題＋送出之內文)')
    else if (isestr(inp.title) && inp.material.split('\n')[0].trim() !== inp.title.trim()) p.push('input.material 須以原文標題為首行(與正式材料同一定義:原文標題＋送出之內文)')
    return p
}

/**
 * domain 之形狀檢核(角色之 domainProblem;抓取套件之 domainNeedsOf——須為物件、須具之函數):納入閘另讀已解析之詞彙表,重標另讀分類清單
 *
 * @param {Array} fns 輸入須具之函數名
 * @param {Object} [opt={}] 輸入 { resolvedVocab:Boolean(vocab 須為 resolveSummarizeVocab 之產物), categories:Boolean(vocab 須含分類清單) }
 * @returns {Function} 回傳 (d) => String(問題說明;空字串＝合格)
 */
const domainNeeds = (fns, opt = {}) => domainNeedsOf(fns, (d) => {
    if (opt.resolvedVocab) {
        try {
            assertSummarizeVocab(d.vocab)
        }
        catch {
            return '之 vocab 須為 resolveSummarizeVocab 之產物（納入閘之判定讀之）'
        }
    }
    if (opt.categories && !(isobj(d.vocab) && isarr(d.vocab.categories) && isestr(d.vocab.fallbackCategory))) return '之 vocab 須含分類清單（categories、fallbackCategory）'
    return ''
})

 *   正規化後之項目合輸出格式(domain.isValidItem)→ 組版(buildNote,不落地)→ 筆記標準(checkNote;不合格或未能判定即不納入)
 *
 * stage 表示停在哪一關:'format'(項目不合輸出格式;正式管線視為未涵蓋)、'gate'(硬性欄位或數字紮根不合格;正式管線不收此項)、
 * 'standards'(已收而組版後之筆記標準不合格;正式管線不落地)、''(納入)。正式管線同一 index 之重複項:前兩關未過者換下一項,
 * 已收者即定(standards 不合格亦不再換)。修補後之重判同正式不再驗格式(o.format:false;項目於修補前已合格式,修補後之標題、
 * 核心知識與概念由閘之 SS2 驗)。
 *
 * @param {Object} item 輸入 AI 回傳之單一知識項目(relevant 為 true;index 為本頁序號)
 * @param {Object} o 輸入設定物件
 * @param {Object} o.domain 輸入摘要 domain(createSummarizeDomain 之產物或相容者;vocab 須為已解析之詞彙表)
 * @param {Object} o.doc 輸入來源文件 { id, title, sourceName, url }
 * @param {String} o.material 輸入材料(原文標題＋送出之內文;SS4 之依據)
 * @param {Integer} o.count 輸入本頁篇數(index 之範圍)
 * @param {Object} [o.chars] 輸入處理範圍 { sourceChars, processedChars }(只進組版之來源章)
 * @param {Boolean} [o.format=true] 輸入是否驗輸出格式(domain.isValidItem;修補後之重判為 false)
 * @returns {Object} 回傳 { ok, stage, std(不納入之主要標準;格式不合為 '格式';納入為 ''), reasons:Array, removed:Array(逐條移除之紀錄), item(正規化後) }
 * @throws {Error} o.domain.vocab 非本套件解析後之詞彙表時拋出
 * @example
 * let d = createSummarizeDomain()
 * let it = { index: 1, relevant: true, title: '甲', summary: '乙', key_points: ['丙'], concepts: ['丁'], claim_type: '教學示範', evidence_level: '中' }
 * console.log(benchAdmitNoteItem(it, { domain: d, doc: { id: 'x', title: '甲', sourceName: '', url: '' }, material: '甲\n丙', count: 1 }).reasons)
 * // => [ '概念 1 個（須 2～6）' ]
 */
export function benchAdmitNoteItem(item, o) {
    const { domain, doc, material, count } = o
    const g = gateNoteItem(item, { vocab: domain.vocab, material, knownConcept: NO_KNOWN_CONCEPT })
    if (o.format !== false && !domain.isValidItem(g.item, count)) {
        return { ok: false, stage: 'format', std: '格式', reasons: ['項目不合輸出格式（index、relevant；知識項須有標題、核心知識與概念）'], removed: [], item: g.item }
    }
    if (!g.ok) return { ok: false, stage: 'gate', std: g.std, reasons: g.reasons, removed: [], item: g.item }
    const n = buildNote({ k: g.item, doc: { publishedAt: '', textFrom: '', ...doc }, domain, notesDir: NOTES_DIR, nowIso: '', ...(isobj(o.chars) ? { chars: o.chars } : {}) })
    const rs = checkNote({ front: n.front, body: n.body, record: n.record }, { vocab: domain.vocab, material })
    const bad = rs.filter((r) => r.status === STATUS.VIOLATION || r.status === STATUS.UNDETERMINED)
    if (bad.length) {
        return { ok: false, stage: 'standards', std: (bad.find((r) => r.status === STATUS.VIOLATION) || bad[0]).std, reasons: bad.map((r) => `${r.std} ${r.detail}`), removed: g.removed, item: g.item }
    }
    return { ok: true, stage: '', std: '', reasons: [], removed: g.removed, item: g.item }
}

/**
 * 摘要之逐案檢核(與 SS5 摘要題組之自帶材料案同形)
 *
 * @param {Object} c 輸入案
 * @returns {Array} 回傳問題說明陣列
 */
function summarizeCaseProblems(c) {
    const inp = isobj(c.input) ? c.input : {}
    const p = []
    if (inp.fixture !== undefined) p.push('以樣本檔定位之案(input.fixture)須先轉為自帶材料 input.material:{ title, sourceName, url, text }')
    else if (!isobj(inp.material)) p.push('缺 input.material:{ title, sourceName, url, text }')
    else {
        const m = inp.material
        if (!isestr(m.title)) p.push('input.material.title 須為非空字串')
        if (!isestr(m.text) || !m.text.trim()) p.push('input.material.text 須為非空字串')
        for (const k of ['sourceName', 'url']) if (typeof m[k] !== 'string') p.push(`input.material.${k} 須為字串(可為空字串)`)
    }
    if (!isobj(c.expect) || typeof c.expect.relevant !== 'boolean') p.push('expect.relevant 須為布林')
    return [...p, ...inputVocabProblems(c)]
}

/**
 * 案之來源文件(提示詞與組版所讀之欄位)
 *
 * @param {Object} c 輸入案
 * @returns {Object} 回傳 { id, title, sourceName, url, text }
 */
function materialDocOf(c) {
    const m = c.input.material
    return { id: c.id, title: m.title, sourceName: m.sourceName, url: m.url, text: m.text }
}

/**
 * 依字數打包之參數(預設同正式摘要)
 *
 * @param {Object} roleOpt 輸入角色參數 { docsPerCall?, charsPerDoc? }
 * @returns {Object} 回傳 { batchSize, charsPerDoc }
 */
const packOptOf = (roleOpt) => ({ batchSize: benchPintOf(roleOpt?.docsPerCall, S.docsPerCall, 'docsPerCall'), charsPerDoc: benchPintOf(roleOpt?.charsPerDoc, S.charsPerDoc, 'charsPerDoc') })

/**
 * 摘要之逐案類別:同一 index 依序取第一個可收之項(正式摘要之部分接受)——判非知識者即收(irrelevant);知識項過格式與閘者即收,
 *   再以組版後之筆記標準定 note-ok／note-bad;皆未收者為 note-bad(判知識而未過閘)
 *
 * @param {*} data 輸入該頁回覆
 * @param {Object} c 輸入該案
 * @param {Object} at 輸入 { index, domain, cases, roleOpt }
 * @returns {Object} 回傳 { cls, valid, problem, fields }
 */
function summarizeVerdictOf(data, c, at) {
    if (!isarr(data)) return noVerdict(`回覆非 JSON 陣列（${typeName(data)}）`, SUMMARIZE_BLANK)
    const its = data.filter((x) => isobj(x) && x.index === at.index)
    if (!its.length) return noVerdict(data.length ? `回覆無 index ${at.index} 之項` : '回覆為空陣列', SUMMARIZE_BLANK)
    const doc = materialDocOf(c)
    // 送出之字數與正式摘要同一規則(packDocsByChars:預算只依本篇字數與每頁字數上限,與同頁他篇無關)
    const budget = packDocsByChars(at.cases.map(materialDocOf), packOptOf(at.roleOpt)).budgets.get(c.id)
    const material = `${doc.title || ''}\n${String(doc.text || '').slice(0, budget)}`
    const chars = { sourceChars: doc.text.length, processedChars: Math.min(doc.text.length, budget) }
    const fieldsOf = (adm) => ({ relevant: true, noteTitle: oneline(textValueOf(adm.item?.title), 60), gate: adm.ok ? '合格' : adm.std, reasons: oneline(adm.reasons.join('；'), 200), removed: adm.removed.length, reason: '' })
    let first = null
    for (const it of its) {
        if (typeof it.relevant !== 'boolean') continue
        if (!it.relevant) return { cls: 'irrelevant', valid: true, problem: '', fields: { ...SUMMARIZE_BLANK, relevant: false, reason: oneline(textValueOf(it.reason), 60) } }
        const adm = benchAdmitNoteItem(it, { domain: at.domain, doc, material, count: at.cases.length, chars })
        if (adm.ok || adm.stage === 'standards') return { cls: adm.ok ? 'note-ok' : 'note-bad', valid: true, problem: '', fields: fieldsOf(adm) }
        if (!first) first = adm
    }
    if (first) return { cls: 'note-bad', valid: first.stage !== 'format', problem: '', fields: fieldsOf(first) }
    return noVerdict(`index ${at.index} 之項無布林之 relevant`, SUMMARIZE_BLANK)
}

/** 修補之無回覆欄位 */
const REPAIR_BLANK = Object.freeze({ problems: '', applied: 0, left: '' })

/**
 * 修補之逐案檢核
 *
 * @param {Object} c 輸入案
 * @returns {Array} 回傳問題說明陣列
 */
function repairCaseProblems(c) {
    const inp = isobj(c.input) ? c.input : {}
    const p = materialProblems(inp)
    if (!isobj(inp.item)) p.push('缺 input.item(有已知問題之筆記項目)')
    else if (!SHAPE_DOMAIN.isValidItem({ ...inp.item, index: 1, relevant: true }, 1)) p.push('input.item 須合輸出格式(標題、核心知識與概念非空;正式管線只對此類項目發修補呼叫)')
    if (!isobj(c.expect) || c.expect.fixed !== true) p.push('expect.fixed 須為 true(題組只收依原文修得好之案)')
    return [...p, ...inputVocabProblems(c)]
}

/**
 * 修補之題目:正規化後之項目與須修補之處(同正式:normalizeNoteItem → repairProblemsOf;提示詞與判定共用)
 *
 * @param {Object} c 輸入案
 * @param {Object} domain 輸入摘要 domain
 * @returns {Object} 回傳 { material, item, problems }
 */
function repairPlanOf(c, domain) {
    const material = String(c.input.material)
    const n0 = normalizeNoteItem({ ...c.input.item, index: 1, relevant: true }, { vocab: domain.vocab, knownConcept: NO_KNOWN_CONCEPT })
    return { material, item: n0.item, problems: repairProblemsOf(n0.item, { vocab: domain.vocab, material }) }
}

/**
 * 修補之逐案類別:applyRepairs 只套回被指出之處 → 同一納入閘;過閘且無逐條移除才算修好
 *
 * @param {*} data 輸入回覆
 * @param {Object} c 輸入該案
 * @param {Object} at 輸入 { domain }
 * @returns {Object} 回傳 { cls, valid, problem, fields }
 */
function repairVerdictOf(data, c, at) {
    if (!checkRepairReply(data)) return noVerdict(`回覆非 {"repairs":[…]} 物件（${typeName(data)}）`, REPAIR_BLANK)
    const plan = repairPlanOf(c, at.domain)
    const a = applyRepairs(plan.item, plan.problems, data)
    const adm = benchAdmitNoteItem(a.item, { domain: at.domain, doc: { id: c.id, title: c.input.title, sourceName: '', url: '' }, material: plan.material, count: 1, format: false })
    const fixed = adm.ok && adm.removed.length === 0
    const left = adm.ok ? adm.removed.map((x) => `移除含原文沒有之數字：${x}`).join('；') : `${adm.std}：${adm.reasons.join('；')}`
    const valid = data.repairs.every((r) => isobj(r) && Number.isInteger(Number(r.no)) && Number(r.no) >= 1 && Number(r.no) <= plan.problems.length && Object.hasOwn(r, 'value'))
    return { cls: fixed ? 'fixed' : 'unfixed', valid, problem: '', fields: { problems: plan.problems.map(fieldPlaceOf).join('、'), applied: a.applied, left: oneline(left, 200) } }
}

    runnerName: 'runSummarizeBench(repair)',
    std: ['SS2', 'SS4'],
    v: 1,
    casesFile: SUMMARIZE_BENCH_CASES_FILES.repair,
    models: SUMMARIZE_BENCH_MODELS,
    caseProblems: repairCaseProblems,
    expectOf: (c) => (c.expect.fixed ? 'fixed' : 'unfixed'),
    caseFieldsOf: (c) => ({ title: c.input.title }),
    domainOf: (vocab) => createSummarizeDomain({ vocab, summarizeCharsPerDoc: S.charsPerDoc }),
    domainProblem: domainNeeds(['buildRepairPrompt', 'isValidItem', 'normalizeQuality', 'renderNoteBody'], { resolvedVocab: true }),
    promptOf: (cases, domain) => {
        const c = cases[0]
        const plan = repairPlanOf(c, domain)
        if (!plan.problems.length) throw new Error(`修補之案 ${c.id} 依現行詞彙與原文無須修補之處（題組須給有已知問題之筆記；正式管線不對此類項目發修補呼叫）`)

/**
 * 缺票之原因(同編號之票皆不合格或互相矛盾)
 *
 * @param {Array} checks 輸入回覆之 checks
 * @param {Integer} index 輸入項之編號
 * @returns {String} 回傳原因
 */
function faithMissingOf(checks, index) {
    const raws = checks.filter((x) => Number(x?.index) === index)
    if (raws.some((x) => checkFaithVerdictOf(x) !== null)) return `同編號 ${index} 之裁決互相矛盾（同正式規則作廢為缺票）`
    const r = raws[0]
    if (!isobj(r)) return `index ${index} 之裁決非物件（缺票）`
    if (String(r.verdict ?? '').trim().toLowerCase() === 'unfaithful') return 'unfaithful 之理由過短或缺（同正式規則視為缺票；須指出筆記哪一欄與原文何處不符）'
    return `裁決「${oneline(String(r.verdict ?? ''), 30)}」非 ${FAITH_VERDICTS.join('／')}（缺票）`
}

/**
 * 忠實性之逐案類別:以正式之單票驗證(seatFaithVotesOf)取該項之票;缺票為無裁決
 *
 * @param {*} data 輸入該頁回覆
 * @param {Object} c 輸入該案
 * @param {Object} at 輸入 { index, cases }
 * @returns {Object} 回傳 { cls, valid, problem, fields }
 */
function faithVerdictOf(data, c, at) {
    if (!isobj(data) || !isarr(data.checks)) return noVerdict(`回覆非 {"checks":[…]} 物件（${typeName(data)}）`, FAITH_BLANK)
    const votes = seatFaithVotesOf(data, new Set(at.cases.map((_, j) => j + 1)))
    if (!votes.has(at.index)) return noVerdict(`回覆無 index ${at.index} 之裁決`, FAITH_BLANK)
    const v = votes.get(at.index)
    if (v === PANEL_UNSURE) return { cls: PANEL_UNSURE, valid: true, problem: '', fields: { reason: '' } }
    if (v === null) return noVerdict(faithMissingOf(data.checks, at.index), FAITH_BLANK)
    return { cls: v.verdict, valid: true, problem: '', fields: { reason: oneline(v.reason, 120) } }
}

/** 重標之無回覆欄位 */
const RETAG_BLANK = Object.freeze({ replied: '' })

/** 重判工作之焦點(新增類別、退役類別、維運排入) */
const RETAG_REASONS = Object.freeze(['add', 'retire', 'requeue'])

/**
 * 重標之逐案檢核(與 SS9 題組同形:焦點在案之頂層);工作範圍同正式(新增類別之工作不含已屬該類者;退役與排入只含原類別＝焦點者)
 *
 * @param {Object} c 輸入案
 * @returns {Array} 回傳問題說明陣列
 */
function retagCaseProblems(c) {
    const p = []
    const f = c.focus
    const okFocus = isobj(f) && RETAG_REASONS.includes(f.reason) && isestr(f.category)
    if (!okFocus) p.push(`focus 須為 { reason:${RETAG_REASONS.join('|')}, category }(本案所屬重判工作之焦點)`)
    const inp = isobj(c.input) ? c.input : {}
    if (!isestr(inp.title)) p.push('缺 input.title')
    if (!isestr(inp.category)) p.push('缺 input.category(筆記之原類別)')
    if (typeof inp.summary !== 'string') p.push('input.summary 須為字串')
    if (inp.points !== undefined && !(isarr(inp.points) && inp.points.every((x) => typeof x === 'string'))) p.push('input.points 須為字串陣列(核心知識)')
    if (!isobj(c.expect) || !isestr(c.expect.category)) p.push('expect.category 須為非空字串')
    if (okFocus && isestr(inp.category)) {
        if (f.reason === 'add' && inp.category === f.category) p.push('新增類別之工作不含已屬該類別之筆記(同正式之重判範圍)')
        if (f.reason !== 'add' && inp.category !== f.category) p.push(`${f.reason} 之工作只含原類別＝焦點類別之筆記(同正式之重判範圍)`)
    }
    return [...p, ...inputVocabProblems(c)]
}

/**
 * 同頁之案與詞彙之一致(於組提示詞時驗,詞彙可由 --vocab 換):焦點類別之存在、應然類別在清單且可達——不合者恆不通過,是題組之缺陷
 *
 * @param {Array} cases 輸入同頁之案(同一焦點)
 * @param {Object} vocab 輸入重標 domain 之詞彙表
 * @returns {Array} 回傳問題說明陣列
 */
function retagPageProblems(cases, vocab) {
    const f = cases[0].focus
    const cats = vocab.categories
    const p = []
    if (f.reason === 'retire' && cats.includes(f.category)) p.push(`焦點 retire 之類別「${f.category}」仍在分類清單`)
    if (f.reason !== 'retire' && !cats.includes(f.category)) p.push(`焦點 ${f.reason} 之類別「${f.category}」不在分類清單`)
    for (const c of cases) {
        const from = c.input.category
        const exp = c.expect.category
        if (!cats.includes(exp)) p.push(`案 ${c.id} 之應然類別「${exp}」不在分類清單`)
        else if (f.reason === 'add' && from !== vocab.fallbackCategory && cats.includes(from) && exp !== from && exp !== f.category) {
            p.push(`案 ${c.id} 之應然「${exp}」不可達（新增類別之工作，原類別在清單且非落點者只在原類別「${from}」與新類別「${f.category}」間擇一）`)
        }
    }
    return p
}

/**
 * 重標之逐案類別:同一 index 取第一個合 isValidItem 之項,回覆類別經 retagTargetOf 落點(同正式重標)
 *
 * @param {*} data 輸入該頁回覆
 * @param {Object} c 輸入該案
 * @param {Object} at 輸入 { index, domain, cases }
 * @returns {Object} 回傳 { cls, valid, problem, fields }
 */
function retagVerdictOf(data, c, at) {
    if (!isarr(data)) return noVerdict(`回覆非 JSON 陣列（${typeName(data)}）`, RETAG_BLANK)
    const it = data.find((x) => x?.index === at.index && at.domain.isValidItem(x, at.cases.length))
    if (!it) {
        const raw = data.find((x) => isobj(x) && x.index === at.index)
        return noVerdict(raw ? `index ${at.index} 之類別「${oneline(typeof raw.category === 'string' ? raw.category : JSON.stringify(raw.category ?? null), 40)}」不在分類清單` : (data.length ? `回覆無 index ${at.index} 之項` : '回覆為空陣列'), RETAG_BLANK)
    }
    const to = retagTargetOf(c.focus, c.input, it.category, taxonomyOf(at.domain.vocab))
    return { cls: to, valid: true, problem: '', fields: { replied: it.category } }
}
